ndvi = ds["ndvi"]

    ndvi = ndvi / 1000.0

ndvi = ndvi.where(ndvi >= -1.0)
ndvi = ndvi.where(ndvi <=  1.0)

# Verificar orden del eje lat
print(float(ds["lat"][0]), float(ds["lat"][-1]))

ndvi = ndvi.sel(
    lat=slice(-3.4, -18.4),   
    lon=slice(-81.0, -70.0)
)

# Extraer lat/lon YA recortados desde el ndvi sliceado
lat = ndvi["lat"]   # ← desde ndvi, no desde ds
lon = ndvi["lon"]   # ← desde ndvi, no desde ds

print("ndvi shape:", ndvi.shape)
print("lat shape:", lat.shape)
print("lon shape:", lon.shape)  

# Graficar
fig = plt.figure(figsize=(10, 10))

    shading="auto",
    cmap="RdYlGn",
